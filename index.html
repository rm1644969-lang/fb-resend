Firebase Admin Sign-In & Anti-Crack Rules (Optional)

Keep Test Mode for fastest setup, OR lock Firestore with the rules below and sign in with an Admin Email/Password created in Firebase Authentication. (Even in Test Mode, every key is HMAC-signed by index.html so nobody can forge a license!) 

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /licenses/{licenseId} {
      allow get: if true;
      allow list, create, delete: if request.auth != null;
      allow update: if request.auth != null
        || (resource.data.status == "active"
            && request.resource.data.diff(resource.data).affectedKeys()
                 .hasOnly(["bound_hwid", "last_seen", "total_checks"])
            && (resource.data.bound_hwid == "" || resource.data.bound_hwid == request.resource.data.bound_hwid));
    }
  }
}
